        AND GCFR.BUSINESS_DATE BETWEEN RT_9BOX.EFF_DATE_KEY AND RT_9BOX.EFF_END_DATE
        AND EXTRACT(YEAR FROM JOB.EFFECTIVE_DATE) = RT_9BOX.EFFECTIVE_YEAR
        
       
LEFT OUTER JOIN SCS01_SEM01_TDE_HR_VR.HRSEM_SFSRI_EMAIL_ADDRESSES EMAIL
   ON EMAIL.EMPLID = JOB.BANK_ID
   AND EMAIL.E_ADDR_TYPE = 'Business'   

LEFT OUTER JOIN (
                    SELECT SUPERVISOR_BANK_ID, COUNT(DISTINCT BANK_ID) AS NO_OF_REPORTEES 
                    FROM (  SELECT * FROM  SCS01_SEM01_TDE_HR_VR.FACT_EMPLOYEEJOB
                            WHERE EMPLOYEE_STATUS IN ('Active','Paid Leave','Unpaid Leave')
                            AND (SELECT BUSINESS_DATE FROM SCS01_SEM01_TDE_HR_VR.GCFR_STREAM_BUSDATE WHERE STREAM_KEY = '4291') BETWEEN EFFECTIVE_DATE AND EFF_END_DATE
                            QUALIFY ROW_NUMBER() OVER( PARTITION BY BANK_ID ORDER BY EFFECTIVE_DATE DESC, EMPLOYEE_RECORD DESC, EFFECTIVE_SEQ DESC) = 1 
                          ) JB_IN
                            GROUP BY SUPERVISOR_BANK_ID
                ) REPORTEES
             ON REPORTEES.SUPERVISOR_BANK_ID = JOB.BANK_ID
   

LEFT OUTER JOIN (
                    SELECT SUPERVISOR_BANK_ID, COUNT(DISTINCT BANK_ID) AS NO_OF_REPORTEES 
                    FROM (  SELECT * FROM  SCS01_SEM01_TDE_HR_VR.FACT_EMPLOYEEJOB
                            WHERE EMPLOYEE_STATUS IN ('Active','Paid Leave','Unpaid Leave')
             --AND EMPLOYMENT_TYPE <> 'Outsourced Worker'
    AND EMPLOYMENT_TYPE NOT IN ('Outsourced Worker','Consultant','NEW Intern', 'Managed Service Worker', 'Managed Outcome Worker')  -- Updated as part of SFEC-24266 fix
                            AND (SELECT BUSINESS_DATE FROM SCS01_SEM01_TDE_HR_VR.GCFR_STREAM_BUSDATE WHERE STREAM_KEY = '4291') BETWEEN EFFECTIVE_DATE AND EFF_END_DATE
                            QUALIFY ROW_NUMBER() OVER( PARTITION BY BANK_ID ORDER BY EFFECTIVE_DATE DESC, EMPLOYEE_RECORD DESC, EFFECTIVE_SEQ DESC) = 1 
                          ) JB_IN
                            GROUP BY SUPERVISOR_BANK_ID
                ) EXCL_OS_RPT
             ON EXCL_OS_RPT.SUPERVISOR_BANK_ID = JOB.BANK_ID

  ON GRADE.GLOBAL_GRADE_DESCR = JOB.GLOBAL_GRADE_CODE

LEFT JOIN SCS01_SEM01_TDE_HR_VR.DIM_GLOBAL_GRD_GRP GRADE_GRP
    ON JOB.GLOBAL_GRADE_CODE = GRADE_GRP.GLOBAL_GRADE_DESCR

  ON PERSONAL_DATA.BANK_ID = JOB.BANK_ID
